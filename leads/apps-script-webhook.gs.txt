TechMonx Leads — Google Sheet webhook + 14-day drip campaign
============================================================

Your website already captures every lead (contact form, chat widget,
booking modal), tags it, logs it to a protected file on the server, and
sends your team an instant email alert plus the visitor an introductory
auto-reply — all of that already works with no setup.

This script adds two extra layers, both running on the free Google
Sheets + Apps Script stack (no paid ESP required):

  1. Every new lead also gets appended as a row to a Google Sheet you
     control ("TechMonx Leads"), so you have a shareable, filterable
     lead list without logging into the server.

  2. Contact-form leads who tick the separate "Also send me occasional
     tips and updates" marketing opt-in checkbox are automatically
     enrolled into a 6-email, 14-day drip sequence: a Day-0 welcome,
     three educational touchpoints (Day 2/5/9) drawing on real
     portfolio case studies and blog posts, and two reminder emails
     (Day 12/14). Every email includes a working unsubscribe link.

  Leads who DON'T tick the marketing checkbox still get the normal
  one-off "thanks for reaching out" auto-reply from contact-handler.php
  — they are never enrolled in the drip.

THIS IS ALREADY LIVE — the sections below are a reference/backup in
case the Apps Script project ever needs to be rebuilt from scratch.

Sheet: "TechMonx Leads"
Apps Script project: "TechMonx Leads Webhook"

SHEET SCHEMA (row 1 headers, columns A–S)
------------------------------------------
A Timestamp | B Form Type | C Name | D Email | E Phone | F Company |
G Service | H Budget | I Timeline | J Tags | K Details | L Notes |
M Marketing Consent | N Unsub Token | O Drip Status | P Drip Stage |
Q Next Send | R Last Sent | S Enrolled Date

Drip Status values: active | completed | unsubscribed
Drip Stage: 0 (Day 0, sent immediately on enrollment) through 5 (Day 14,
the final reminder). After stage 5 sends, Drip Status flips to
"completed" automatically.

REBUILDING FROM SCRATCH
-------------------------
1. Go to https://sheets.google.com, create a blank spreadsheet, rename
   it "TechMonx Leads", and add the headers above to row 1.
2. Extensions > Apps Script. Delete any starter code and paste the full
   script below. Save, name the project "TechMonx Leads Webhook".
3. Run the "installDailyTrigger" function once from the editor (select
   it in the function dropdown, click Run). Google will prompt for
   authorization the first time — click through and Allow. This installs
   a daily 9am trigger that sends whichever drip stage is due for each
   active lead.
4. The first time a real lead enrolls and the script tries to send an
   email, it will also need to authorize the Gmail-send scope
   (script.send_mail) — this happens automatically the first time
   MailApp.sendEmail actually runs; if the very first send fails with a
   permissions error, open Executions in the left sidebar, find the
   failed run, and click "Click here to provide permissions".
5. Deploy > Manage deployments > pencil/edit icon on the existing
   deployment > Version: "New version" > Deploy. This keeps the same
   /exec URL. (Use "New deployment" only if starting completely fresh —
   that would generate a different URL, which would then need to be
   re-pasted into $crmWebhookUrl in contact-handler.php.)
6. Copy the Web app URL (ends in /exec) into contact-handler.php:
     $crmWebhookUrl = 'PASTE_YOUR_URL_HERE';

// ============================================================
// TechMonx Leads Webhook + Drip Campaign Engine
// ============================================================

var SHEET_NAME = 'Sheet1';

var COL = {
  TIMESTAMP: 1,
  FORM_TYPE: 2,
  NAME: 3,
  EMAIL: 4,
  PHONE: 5,
  COMPANY: 6,
  SERVICE: 7,
  BUDGET: 8,
  TIMELINE: 9,
  TAGS: 10,
  DETAILS: 11,
  NOTES: 12,
  MARKETING_CONSENT: 13,
  UNSUB_TOKEN: 14,
  DRIP_STATUS: 15,
  DRIP_STAGE: 16,
  NEXT_SEND: 17,
  LAST_SENT: 18,
  ENROLLED_DATE: 19
};

var DRIP_OFFSETS = [0, 2, 5, 9, 12, 14];

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME)
    || SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);

  var tags = Array.isArray(data.tags) ? data.tags.join(', ') : '';
  var name = data.name || '';
  var email = data.email || '';
  var service = data.service_label || data.service || '';
  var marketingConsent = data.marketing_consent === true || data.marketing_consent === 'true' || data.marketing_consent === 'on';

  var row = [
    data.timestamp || new Date().toISOString(),
    data.form_type || '',
    name,
    email,
    data.phone || '',
    data.company || '',
    service,
    data.budget || '',
    data.timeline || '',
    tags,
    data.details || '',
    data.notes || data.message || '',
    marketingConsent ? 'TRUE' : 'FALSE',
    '',
    '',
    '',
    '',
    '',
    ''
  ];

  sheet.appendRow(row);
  var rowNum = sheet.getLastRow();

  if (marketingConsent && email) {
    enrollInDrip_(sheet, rowNum, name, email, service);
  }

  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

function enrollInDrip_(sheet, rowNum, name, email, service) {
  var token = Utilities.getUuid();
  var now = new Date();

  sheet.getRange(rowNum, COL.UNSUB_TOKEN).setValue(token);
  sheet.getRange(rowNum, COL.DRIP_STATUS).setValue('active');
  sheet.getRange(rowNum, COL.ENROLLED_DATE).setValue(now);

  sendDripEmail_(0, name, email, service, token);

  sheet.getRange(rowNum, COL.DRIP_STAGE).setValue(1);
  sheet.getRange(rowNum, COL.LAST_SENT).setValue(now);
  sheet.getRange(rowNum, COL.NEXT_SEND).setValue(addDays_(now, DRIP_OFFSETS[1]));
}

function addDays_(date, days) {
  var d = new Date(date);
  d.setDate(d.getDate() + days);
  return d;
}

function sendDripEmails() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  var range = sheet.getRange(2, 1, lastRow - 1, COL.ENROLLED_DATE);
  var values = range.getValues();
  var now = new Date();

  for (var i = 0; i < values.length; i++) {
    var r = values[i];
    var status = r[COL.DRIP_STATUS - 1];
    var stage = r[COL.DRIP_STAGE - 1];
    var nextSend = r[COL.NEXT_SEND - 1];

    if (status !== 'active') continue;
    if (!nextSend || new Date(nextSend) > now) continue;

    var rowNum = i + 2;
    var name = r[COL.NAME - 1];
    var email = r[COL.EMAIL - 1];
    var service = r[COL.SERVICE - 1];
    var token = r[COL.UNSUB_TOKEN - 1];

    sendDripEmail_(stage, name, email, service, token);

    var nextStage = stage + 1;
    sheet.getRange(rowNum, COL.LAST_SENT).setValue(now);

    if (nextStage >= DRIP_OFFSETS.length) {
      sheet.getRange(rowNum, COL.DRIP_STATUS).setValue('completed');
      sheet.getRange(rowNum, COL.NEXT_SEND).setValue('');
    } else {
      sheet.getRange(rowNum, COL.DRIP_STAGE).setValue(nextStage);
      var enrolled = r[COL.ENROLLED_DATE - 1] || now;
      sheet.getRange(rowNum, COL.NEXT_SEND).setValue(addDays_(enrolled, DRIP_OFFSETS[nextStage]));
    }
  }
}

function installDailyTrigger() {
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === 'sendDripEmails') {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }
  ScriptApp.newTrigger('sendDripEmails')
    .timeBased()
    .everyDays(1)
    .atHour(9)
    .create();
}

function doGet(e) {
  var token = e.parameter.token;
  var html;

  if (!token) {
    html = '<html><body style="font-family:sans-serif;padding:40px;text-align:center;"><h2>Invalid unsubscribe link</h2></body></html>';
    return HtmlService.createHtmlOutput(html);
  }

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  var lastRow = sheet.getLastRow();
  var found = false;

  if (lastRow >= 2) {
    var tokens = sheet.getRange(2, COL.UNSUB_TOKEN, lastRow - 1, 1).getValues();
    for (var i = 0; i < tokens.length; i++) {
      if (tokens[i][0] === token) {
        var rowNum = i + 2;
        sheet.getRange(rowNum, COL.DRIP_STATUS).setValue('unsubscribed');
        sheet.getRange(rowNum, COL.NEXT_SEND).setValue('');
        found = true;
        break;
      }
    }
  }

  html = found
    ? '<html><body style="font-family:sans-serif;padding:40px;text-align:center;"><h2>You have been unsubscribed</h2><p>You will no longer receive these emails from TechMonx.</p></body></html>'
    : '<html><body style="font-family:sans-serif;padding:40px;text-align:center;"><h2>Link not found</h2><p>This unsubscribe link is no longer valid.</p></body></html>';

  return HtmlService.createHtmlOutput(html);
}

function sendDripEmail_(stage, name, email, service, token) {
  if (!email) return;
  var firstName = (name || '').split(' ')[0] || 'there';
  var svc = service || 'your project';
  var unsubscribeUrl = ScriptApp.getService().getUrl() + '?token=' + token;
  var tpl = getEmailTemplate_(stage, firstName, svc, unsubscribeUrl);
  if (!tpl) return;

  MailApp.sendEmail({
    to: email,
    subject: tpl.subject,
    htmlBody: tpl.body,
    name: 'TechMonx'
  });
}

That's it — no ongoing cost, no third-party ESP account. Gmail's daily
send quota for a regular account (~100-500 emails/day) applies, which is
comfortably enough for a small business's lead volume. If TechMonx ever
outgrows that, swap MailApp.sendEmail for a dedicated ESP later without
touching the rest of the pipeline.
